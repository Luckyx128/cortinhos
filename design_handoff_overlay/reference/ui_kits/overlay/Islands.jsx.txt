/* ── busca / command palette ─────────────────────────────────── */
function SearchIsland({ query, setQuery, results }) {
  return (
    <div style={{ display: 'flex', flexDirection: 'column' }}>
      <div style={{ display: 'flex', alignItems: 'center', gap: 10, height: 44, padding: '0 4px' }}>
        <Icon name="search" size={18} color="var(--text-tertiary)" />
        <input
          value={query} onChange={e => setQuery(e.target.value)}
          placeholder="Buscar atalho, pasta ou comando…"
          style={{
            flex: 1, minWidth: 0, background: 'transparent', border: 'none', outline: 'none',
            fontFamily: 'var(--font-body)', fontSize: 'var(--fs-subtitle)', color: 'var(--text-primary)',
          }} />
        <span style={{ fontFamily: 'var(--font-mono)', fontSize: 'var(--fs-caption)', color: 'var(--text-tertiary)', background: 'var(--surface-2)', border: '1px solid var(--border-subtle)', borderRadius: 'var(--radius-sm)', padding: '3px 7px' }}>Enter</span>
      </div>
      {query && (
        <div style={{ display: 'flex', flexDirection: 'column', gap: 2, borderTop: '1px solid var(--border-subtle)', paddingTop: 6, marginTop: 2 }}>
          {results.length === 0 && <div style={{ padding: '8px 6px', fontFamily: 'var(--font-body)', fontSize: 'var(--fs-body)', color: 'var(--text-tertiary)' }}>Nada encontrado.</div>}
          {results.slice(0, 4).map((r, i) => (
            <div key={r.id} style={{ display: 'flex', alignItems: 'center', gap: 9, padding: '7px 8px', borderRadius: 'var(--radius-sm)', background: i === 0 ? 'var(--accent-soft)' : 'transparent' }}>
              <Icon name={r.type} size={16} color="var(--accent)" />
              <span style={{ flex: 1, fontFamily: 'var(--font-body)', fontSize: 'var(--fs-body)', color: 'var(--text-primary)' }}>{r.name}</span>
              {r.hotkey && <span style={{ fontFamily: 'var(--font-mono)', fontSize: 'var(--fs-caption)', color: 'var(--text-tertiary)' }}>{r.hotkey}</span>}
            </div>
          ))}
        </div>
      )}
    </div>
  );
}

/* ── grid de atalhos ─────────────────────────────────────────── */
function GridIsland({ items, mode, onOpen }) {
  if (mode === 'list') {
    return (
      <div style={{ display: 'flex', flexDirection: 'column', height: '100%', minHeight: 0 }}>
        <IslandHead title="Atalhos" count={items.length} />
        <div style={{ display: 'flex', flexDirection: 'column', gap: 2, overflow: 'auto', minHeight: 0 }}>
          {items.map(it => <ListRow key={it.id} item={it} onOpen={onOpen} />)}
        </div>
      </div>
    );
  }
  const cols = mode === 'tiles5' ? 5 : 7;
  return (
    <div style={{ display: 'flex', flexDirection: 'column', height: '100%', minHeight: 0 }}>
      <IslandHead title="Atalhos" count={items.length} />
      <div style={{ display: 'grid', gridTemplateColumns: `repeat(${cols}, minmax(0, 1fr))`, gap: 10, alignContent: 'start', overflow: 'auto', minHeight: 0, paddingRight: 2 }}>
        {items.map(it => (
          <ShortcutTile key={it.id} name={it.name} type={it.type} hotkey={it.hotkey} style={{ minWidth: 0 }}
            running={it.running} pinned={it.pinned} onOpen={() => onOpen(it)} />
        ))}
      </div>
    </div>
  );
}

function ListRow({ item, onOpen }) {
  const [hover, setHover] = React.useState(false);
  return (
    <button type="button" onClick={() => onOpen(item)}
      onMouseEnter={() => setHover(true)} onMouseLeave={() => setHover(false)}
      style={{
        display: 'flex', alignItems: 'center', gap: 9, width: '100%', height: 34, padding: '0 8px',
        border: 'none', borderRadius: 'var(--radius-sm)', cursor: 'pointer', textAlign: 'left',
        background: hover ? 'var(--taskbar-hover)' : 'transparent',
      }}>
      <Icon name={item.type} size={16} color="var(--accent)" />
      <span style={{ flex: 1, minWidth: 0, fontFamily: 'var(--font-body)', fontSize: 'var(--fs-body)', color: 'var(--text-primary)', overflow: 'hidden', textOverflow: 'ellipsis', whiteSpace: 'nowrap' }}>{item.name}</span>
      {item.running && <span style={{ width: 6, height: 6, borderRadius: '50%', background: 'var(--success)' }} />}
      {item.hotkey && <span style={{ fontFamily: 'var(--font-mono)', fontSize: 'var(--fs-caption)', color: 'var(--text-tertiary)' }}>{item.hotkey.replace('Ctrl+Alt+', '⌃⌥')}</span>}
    </button>
  );
}

function IslandHead({ title, count, action }) {
  return (
    <div style={{ display: 'flex', alignItems: 'baseline', justifyContent: 'space-between', marginBottom: 10 }}>
      <span style={islandLabel}>{title}{count != null && <span style={{ color: 'var(--text-tertiary)', opacity: .7 }}> · {count}</span>}</span>
      {action}
    </div>
  );
}

/* ── mic + áudio ─────────────────────────────────────────────── */
function MicIsland({ muted, onToggle, deaf, onToggleDeaf }) {
  return (
    <div style={{ display: 'flex', alignItems: 'center', gap: 8 }}>
      <PillBtn active={muted} danger onClick={onToggle} icon={muted ? 'mic-off' : 'mic'}
        label={muted ? 'Mutado' : 'Mic'} />
      <PillBtn active={deaf} danger onClick={onToggleDeaf} icon="notification"
        label={deaf ? 'Surdo' : 'Áudio'} />
      <div style={{ flex: 1 }} />
      <div style={{ display: 'flex', alignItems: 'flex-end', gap: 2, height: 22 }}>
        {[0.35, 0.7, 1, 0.55, 0.2].map((v, i) => (
          <span key={i} style={{
            width: 3, borderRadius: 2, height: `${(muted ? 0.12 : v) * 100}%`,
            background: muted ? 'var(--text-tertiary)' : 'var(--success)',
            transition: 'height var(--dur-med) var(--ease-out)',
          }} />
        ))}
      </div>
    </div>
  );
}

function PillBtn({ icon, label, active, danger, onClick }) {
  const [hover, setHover] = React.useState(false);
  const tone = active && danger ? 'var(--danger)' : 'var(--text-primary)';
  return (
    <button type="button" onClick={onClick}
      onMouseEnter={() => setHover(true)} onMouseLeave={() => setHover(false)}
      style={{
        display: 'flex', alignItems: 'center', gap: 7, height: 34, padding: '0 11px',
        borderRadius: 'var(--radius-sm)', cursor: 'pointer', color: tone,
        background: active && danger ? 'var(--danger-soft)' : hover ? 'var(--taskbar-hover)' : 'var(--surface-2)',
        border: `1px solid ${active && danger ? 'var(--danger)' : 'var(--border-subtle)'}`,
        fontFamily: 'var(--font-body)', fontSize: 'var(--fs-body)', fontWeight: 'var(--fw-medium)',
      }}>
      <span style={{ display: 'inline-flex', width: 16, height: 16 }}><Icon name={icon} size="100%" /></span>
      {label}
    </button>
  );
}

/* ── performance ─────────────────────────────────────────────── */
function PerfIsland({ perf }) {
  return (
    <div>
      <IslandHead title="Desempenho" />
      <div style={{ display: 'flex', gap: 6 }}>
        {[['CPU', perf.cpu, '%'], ['GPU', perf.gpu, '%'], ['FPS', perf.fps, '']].map(([k, v, unit]) => (
          <div key={k} style={{ ...islandCard, flex: 1, padding: '7px 9px' }}>
            <div style={{ fontFamily: 'var(--font-mono)', fontSize: 9.5, color: 'var(--text-tertiary)', letterSpacing: '.06em' }}>{k}</div>
            <div style={{ fontFamily: 'var(--font-display)', fontSize: 17, fontWeight: 'var(--fw-semibold)', color: v > 85 && unit === '%' ? 'var(--danger)' : 'var(--text-primary)', lineHeight: 1.2 }}>{v}{unit}</div>
            {unit === '%' && (
              <div style={{ height: 2, background: 'var(--surface-3)', borderRadius: 1, marginTop: 4 }}>
                <div style={{ height: '100%', width: `${v}%`, background: v > 85 ? 'var(--danger)' : 'var(--accent)', borderRadius: 1, transition: 'width var(--dur-med) var(--ease-out)' }} />
              </div>
            )}
          </div>
        ))}
      </div>
    </div>
  );
}

/* ── notificações ────────────────────────────────────────────── */
function NotifIsland({ items, onClear }) {
  return (
    <div style={{ display: 'flex', flexDirection: 'column', height: '100%', minHeight: 0 }}>
      <IslandHead title="Notificações" count={items.length} action={

      } />
      <div style={{ display: 'flex', flexDirection: 'column', gap: 4, overflow: 'auto', minHeight: 0 }}>
        {items.length === 0 && <div style={{ fontFamily: 'var(--font-body)', fontSize: 'var(--fs-body)', color: 'var(--text-tertiary)' }}>Sem notificações.</div>}

            <Icon name={n.icon || 'app'} size={15} color="var(--accent)" style={{ marginTop: 1, flexShrink: 0 }} />
