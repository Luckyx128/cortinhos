/* ── island wrapper: chrome, drag, edit affordances ───────────── */
function Island({ id, label, pos, size, edit, scrim, onDrag, bare, children }) {
  const [drag, setDrag] = React.useState(false);
  const start = React.useRef(null);

  const down = (e) => {
    if (!edit) return;
    e.preventDefault(); e.stopPropagation();
    start.current = { px: e.clientX, py: e.clientY, x: pos.x, y: pos.y };
    setDrag(true);
    e.currentTarget.setPointerCapture(e.pointerId);
  };
  const move = (e) => {
    if (!drag || !start.current) return;
    const k = window.__overlayScale || 1;
    const nx = start.current.x + (e.clientX - start.current.px) / k;
    const ny = start.current.y + (e.clientY - start.current.py) / k;
    onDrag(id, snap(nx, ny, size));
  };
  const up = (e) => { setDrag(false); start.current = null; try { e.currentTarget.releasePointerCapture(e.pointerId); } catch (_) {} };

  return (
    <div
      data-island={id}
      onPointerDown={down} onPointerMove={move} onPointerUp={up} onPointerCancel={up}
      style={{
        position: 'absolute', left: pos.x, top: pos.y, width: size.w,
        height: size.h || 'auto', boxSizing: 'border-box',
        zIndex: drag ? 40 : 20,
        cursor: edit ? (drag ? 'grabbing' : 'grab') : 'default',
        transform: drag ? 'scale(1.02)' : 'scale(1)',
        transition: drag ? 'none' : 'left var(--dur-med) var(--ease-spring), top var(--dur-med) var(--ease-spring), width var(--dur-med) var(--ease-spring), height var(--dur-med) var(--ease-spring)',
      }}>
      {scrim === 'halo' && (
        <div style={{
          position: 'absolute', inset: -100, zIndex: -1, pointerEvents: 'none',
          background: 'radial-gradient(ellipse at center, rgba(6,8,11,.85) 0%, rgba(6,8,11,.66) 42%, rgba(6,8,11,0) 74%)',
        }} />
      )}
      {edit && (
        <div style={{
          position: 'absolute', inset: -8, borderRadius: 'calc(var(--radius-lg) + 6px)',
          border: `1.5px dashed ${drag ? 'var(--accent)' : 'var(--border-strong)'}`,
          pointerEvents: 'none',
        }} />
      )}
      {edit && (
        <div style={{
          position: 'absolute', left: -8, top: -34, display: 'flex', alignItems: 'center', gap: 4,
          padding: '3px 4px 3px 8px', borderRadius: 'var(--radius-sm)',
          background: 'var(--surface-3)', border: '1px solid var(--border-subtle)', whiteSpace: 'nowrap',
        }}>
          <span style={{ fontFamily: 'var(--font-body)', fontSize: 'var(--fs-caption)', fontWeight: 'var(--fw-semibold)', color: 'var(--text-primary)' }}>{label}</span>
          <span style={{ display: 'inline-flex', width: 20, height: 20, alignItems: 'center', justifyContent: 'center', color: 'var(--text-tertiary)' }}><Icon name="settings" size={13} /></span>
          <span style={{ display: 'inline-flex', width: 20, height: 20, alignItems: 'center', justifyContent: 'center', borderRadius: 4, background: 'var(--accent)', color: 'var(--text-on-accent)' }}><Icon name="pin" size={12} /></span>
        </div>
      )}
      {bare ? children : (
        <div style={{
          height: '100%', boxSizing: 'border-box', padding: 16,
          background: 'var(--material-acrylic)', backdropFilter: 'var(--material-acrylic-blur)',
          border: '1px solid var(--border-subtle)', borderRadius: 'var(--radius-lg)',
          boxShadow: 'var(--elev-4)', color: 'var(--text-primary)',
          display: 'flex', flexDirection: 'column', minHeight: 0,
        }}>{children}</div>
      )}
    </div>
  );
}

function snap(x, y, size) {
  const M = 24, w = size.w, h = size.h || 90;
  const cands = [[x, M], [x, STAGE_H - h - M], [M, y], [STAGE_W - w - M, y], [(STAGE_W - w) / 2, y], [x, 0]];
  let nx = x, ny = y;
  for (const [cx, cy] of cands) {
    if (Math.abs(cx - x) < 14) nx = cx;
    if (Math.abs(cy - y) < 14) ny = cy;
  }
  nx = Math.max(0, Math.min(STAGE_W - w, Math.round(nx)));
  ny = Math.max(0, Math.min(STAGE_H - h, Math.round(ny)));
  return { x: nx, y: ny };
}

  const [perf, setPerf] = React.useState({ cpu: 41, gpu: 88, fps: 144 });
  const wrapRef = React.useRef(null);
  const [scale, setScale] = React.useState(1);

  const P = PRESETS[preset];
  const base = { ...P.pos, ...(notchMode === 'ilha' ? (P.posIsland || {}) : {}) };

  const sizeFor = (id) => ({ ...SIZES[id], ...((P.size || {})[id] || {}) });

  React.useEffect(() => {
    const el = wrapRef.current; if (!el) return;
    const fit = () => {
      const k = Math.min(el.clientWidth / STAGE_W, el.clientHeight / STAGE_H);
      window.__overlayScale = k; setScale(k);
    };
    fit();
    const ro = new ResizeObserver(fit); ro.observe(el);
    return () => ro.disconnect();
  }, []);

  React.useEffect(() => {
    localStorage.setItem(STORE_KEY, JSON.stringify({ preset, layout, scrim, notchMode }));

  React.useEffect(() => {
    const onKey = (e) => {
      const typing = e.target && /input|textarea/i.test(e.target.tagName);
      if (e.key === 'Escape') { setFeas(false); setOpen(false); return; }
      if (typing) return;
      const k = e.key.toLowerCase();
      if (k === 'l') setEdit(v => !v);
      if (k === 'v') setFeas(v => !v);
      if (k === 'h') setRail(v => !v);
      if (k === 'o') setOpen(v => !v);
    };

            <span style={{ fontFamily: 'var(--font-mono)', fontSize: 10, letterSpacing: '.14em', textTransform: 'uppercase', color: 'rgba(255,255,255,.4)' }}>exploração</span>
            {preset === 'A' && scrim === 'vinheta' && notchMode === 'separado' && (
              <span style={{ display: 'inline-flex', alignItems: 'center', gap: 6, padding: '3px 9px', borderRadius: 'var(--radius-pill)', background: 'var(--success-soft)', border: '1px solid var(--success)', fontFamily: 'var(--font-mono)', fontSize: 10, color: 'var(--success)' }}>direção escolhida</span>
            )}
            <RailRow label="Arranjo"><Seg value={preset} onChange={usePreset} options={[{ v: 'A', l: 'A' }, { v: 'B', l: 'B' }, { v: 'C', l: 'C' }]} /></RailRow>
            <RailRow label="Escurecer"><Seg value={scrim} onChange={setScrim} options={[{ v: 'uniforme', l: 'unif' }, { v: 'vinheta', l: 'vinh' }, { v: 'halo', l: 'halo' }, { v: 'nenhum', l: 'off' }]} /></RailRow>
            <RailRow label="Notch"><Seg value={notchMode} onChange={setNotchMode} options={[{ v: 'ilha', l: 'ilha-mãe' }, { v: 'separado', l: 'sep' }, { v: 'oculto', l: 'off' }]} /></RailRow>
            <div style={{ display: 'flex', gap: 6, flexWrap: 'wrap' }}>
              <button type="button" onClick={() => setEdit(v => !v)} style={edit ? activeBtn : ghostBtn}>L · editar layout</button>
              <button type="button" onClick={() => setHitViz(v => !v)} style={hitViz ? activeBtn : ghostBtn}>click-through</button>
              <button type="button" onClick={() => setOpen(v => !v)} style={ghostBtn}>O · {open ? 'fechar' : 'abrir'}</button>
              <button type="button" onClick={() => setFeas(true)} style={ghostBtn}>V · viabilidade</button>
            </div>
            <span style={{ flex: 1 }} />
            <button type="button" onClick={() => setRail(false)} style={{ ...ghostBtn, padding: '3px 8px', fontSize: 10 }}>H ocultar</button>
          </div>
          <div style={{ fontFamily: 'var(--font-body)', fontSize: 11.5, lineHeight: 1.5, color: 'rgba(255,255,255,.5)', textWrap: 'pretty' }}>
            <b style={{ color: 'rgba(255,255,255,.82)' }}>{P.name}</b> — {P.blurb} <span style={{ opacity: .45 }}>·</span> <b style={{ color: 'rgba(255,255,255,.82)' }}>{SCRIMS[scrim].label}</b> — {SCRIMS[scrim].desc} <span style={{ opacity: .45 }}>·</span> <b style={{ color: 'rgba(255,255,255,.82)' }}>{NOTCH_MODES[notchMode].label}</b> — {NOTCH_MODES[notchMode].desc}
          </div>
        </div>
      ) : (
        <button type="button" onClick={() => setRail(true)} style={{ ...ghostBtn, position: 'fixed', left: 16, bottom: 16, zIndex: 100 }}>exploração</button>
      )}

      {feas && <window.FeasibilityPanel onClose={() => setFeas(false)} />}
    </div>
  );
}

const ghostBtn = {
  padding: '5px 9px', borderRadius: 5, cursor: 'pointer',
  background: 'rgba(255,255,255,.06)', border: '1px solid rgba(255,255,255,.16)',
  color: 'rgba(255,255,255,.78)', fontFamily: 'var(--font-mono)', fontSize: 11,
};
const activeBtn = { ...ghostBtn, background: 'var(--accent)', color: 'var(--text-on-accent)', border: '1px solid transparent' };
